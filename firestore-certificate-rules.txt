// =========================================================
// CWS ACADEMY
// CERTIFICATE + PROFESSIONAL CREDENTIAL VERIFICATION RULES
// =========================================================

match /certificateVerifications/{credentialId} {

  // Public verification for employers and recruiters.
  allow read: if true;


  // Current browser-based issuance:
  // authenticated users may create only a credential record
  // associated with the same credentialId and an active status.
  //
  // Course certificate records may use `userId`.
  // Career-path public records intentionally omit uid from the
  // public document, so ownership is enforced by authenticated
  // issuance flow rather than public fields in this interim model.
  allow create:
      if request.auth != null
      && request.resource.data.credentialId == credentialId
      && request.resource.data.status == "active";


  // Allow authenticated refresh of an existing verification record
  // only while keeping the same credential ID.
  //
  // Production recommendation: move this to a trusted server
  // function and set update/delete to false.
  allow update:
      if request.auth != null
      && request.resource.data.credentialId == credentialId
      && resource.data.credentialId == credentialId;


  allow delete:
      if false;
}
