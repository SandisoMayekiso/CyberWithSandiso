// IMPORTANT:

// =========================================================
// PUBLIC COURSE CERTIFICATE VERIFICATION
// =========================================================

match /certificateVerifications/{credentialId} {

  // Employers and recruiters can publicly verify credentials.
  allow read: if true;


  // Interim client-side certificate issuance.
  //
  // A signed-in student may create a verification document
  // only when:
  // - the document belongs to their own Firebase uid
  // - the credentialId inside the document matches the
  //   document path
  // - the credential is initially created as active
  allow create:
      if request.auth != null
      && request.resource.data.userId == request.auth.uid
      && request.resource.data.credentialId == credentialId
      && request.resource.data.status == "active";


  // Once issued, students cannot modify or delete the
  // public verification record.
  allow update, delete:
      if false;
}


// =========================================================
// STUDENT CAPSTONE PROGRESS
// =========================================================

match /users/{userId}/capstones/{capstoneId} {

  // A signed-in student may read only their own
  // capstone progress and grading record.
  allow read:
      if request.auth != null
      && request.auth.uid == userId;


  // Current development implementation:
  // students may create their own capstone progress record.
  allow create:
      if request.auth != null
      && request.auth.uid == userId;


  // Current development implementation:
  // students may update their own capstone progress.
  //
  // NOTE:
  // Before production, fields such as:
  //
  // passed
  // score
  // grading
  // reviewStatus
  // passedAt
  //
  // should ideally be written by a trusted server-side
  // function rather than directly from the browser.
  allow update:
      if request.auth != null
      && request.auth.uid == userId;


  // Prevent students from deleting capstone history.
  allow delete:
      if false;
}


// =========================================================
// CAREER PATH CERTIFICATES
// =========================================================

match /users/{userId}/careerPathCertificates/{pathId} {

  // Students may read only their own Career Path Certificates.
  allow read:
      if request.auth != null
      && request.auth.uid == userId;


  // Current development implementation:
  // the browser may create a Career Path Certificate only
  // inside the signed-in student's own account.
  allow create:
      if request.auth != null
      && request.auth.uid == userId;


  // Allow the student's own Career Path Certificate record
  // to be updated during the current development stage.
  //
  // Production recommendation:
  // move credential issuance and modification to a trusted
  // Cloud Function / Netlify Function.
  allow update:
      if request.auth != null
      && request.auth.uid == userId;


  // Prevent browser-side deletion of issued credentials.
  allow delete:
      if false;
}
